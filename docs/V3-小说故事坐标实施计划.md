```ts
export type NarrativePerson = 'first' | 'third_limited' | 'third_omniscient'
export type TimelineSource = 'architecture' | 'chapter' | 'manual'
export type TimelineStatus = 'planned' | 'confirmed'

export const NARRATIVE_PERSONS: NarrativePerson[] = ['first', 'third_limited', 'third_omniscient']
export const DEFAULT_NARRATIVE_PERSON: NarrativePerson = 'first'
export const LEGACY_NARRATIVE_PERSON: NarrativePerson = 'third_limited'

export const NARRATIVE_PERSON_LABELS: Record<NarrativePerson, string> = {
  first: '第一人称',
  third_limited: '第三人称限知',
  third_omniscient: '第三人称全知',
}

export const NARRATIVE_PERSON_RULES: Record<NarrativePerson, string> = {
  first: '第一人称（「我」）叙述。只写视角人物能看到、听到、知道和想到的内容；不进入其他角色内心，不写视角人物不在场的场景。',
  third_limited: '第三人称限知叙述。围绕视角人物展开，只呈现视角人物可知的信息；不进入其他角色内心。',
  third_omniscient: '第三人称全知叙述。可进入不同角色的内心并交代全局信息，但不得破坏章节悬念与叙事连贯。',
}

export class StoryAtlasValidationError extends Error {
  constructor(message: string) {
    super(message)
    this.name = 'StoryAtlasValidationError'
  }
}

export function normalizeNarrativePerson(
  value: unknown,
  fallback: NarrativePerson = LEGACY_NARRATIVE_PERSON,
): NarrativePerson {
  return typeof value === 'string' && (NARRATIVE_PERSONS as string[]).includes(value)
    ? value as NarrativePerson
    : fallback
}

export function narrativePersonLabel(person: NarrativePerson): string {
  return NARRATIVE_PERSON_LABELS[person]
}

export interface TimelineEvent {
  id: string
  chapterNo: number
  timeLabel: string
  title: string
  summary: string
  placeId?: string
  placeName?: string
  characters?: string[]
  importance: 1 | 2 | 3 | 4 | 5
  source: TimelineSource
  status: TimelineStatus
}

export interface NormalizeTimelineOptions {
  maxChapter?: number
  strict?: boolean
  defaultSource?: TimelineSource
  defaultStatus?: TimelineStatus
  idPrefix?: string
  placeIds?: Iterable<string>
  fixedChapterNo?: number
  maxEvents?: number
}

const TIMELINE_ID = /^[A-Za-z0-9_-]{1,64}$/
const TIMELINE_SOURCES: TimelineSource[] = ['architecture', 'chapter', 'manual']
const TIMELINE_STATUSES: TimelineStatus[] = ['planned', 'confirmed']

function cleanText(value: unknown, limit: number): string {
  return typeof value === 'string' ? value.replace(/\s+/g, ' ').trim().slice(0, limit) : ''
}

function clampImportance(value: unknown): 1 | 2 | 3 | 4 | 5 {
  const n = Math.round(Number(value))
  return (Number.isFinite(n) ? Math.max(1, Math.min(5, n)) : 3) as 1 | 2 | 3 | 4 | 5
}

function parseInput(input: unknown, strict: boolean): unknown[] {
  if (input === null || input === undefined) return []
  if (typeof input === 'string') {
    try {
      input = JSON.parse(input)
    } catch {
      if (strict) throw new StoryAtlasValidationError('时间线 JSON 格式不正确')
      return []
    }
  }
  if (!Array.isArray(input)) {
    if (strict) throw new StoryAtlasValidationError('timeline_events 必须是数组')
    return []
  }
  return input
}

export function normalizeTimelineEvents(
  input: unknown,
  options: NormalizeTimelineOptions = {},
): { events: TimelineEvent[]; warnings: string[] } {
  const {
    maxChapter = Number.MAX_SAFE_INTEGER,
    strict = false,
    defaultSource = 'manual',
    defaultStatus = 'planned',
    idPrefix = 'evt',
    fixedChapterNo,
  } = options
  const maxEvents = Math.max(1, options.maxEvents ?? 2000)
  const placeIds = options.placeIds ? new Set(options.placeIds) : null
  const warnings: string[] = []
  const events: TimelineEvent[] = []
  const usedIds = new Set<string>()

  const rawList = parseInput(input, strict)
  if (rawList.length > maxEvents) {
    if (strict) throw new StoryAtlasValidationError(`时间线事件不能超过 ${maxEvents} 条`)
    warnings.push(`时间线事件超过 ${maxEvents} 条，已截断`)
  }

  for (const raw of rawList.slice(0, maxEvents)) {
    if (!raw || typeof raw !== 'object' || Array.isArray(raw)) {
      if (strict) throw new StoryAtlasValidationError('时间线事件必须是对象')
      warnings.push('丢弃非对象时间线事件')
      continue
    }
    const item = raw as Record<string, unknown>
    const chapterNo = fixedChapterNo ?? Math.round(Number(item.chapterNo))
    if (!Number.isInteger(chapterNo) || chapterNo < 1 || chapterNo > maxChapter) {
      if (strict) throw new StoryAtlasValidationError(`时间线章节号必须在 1..${maxChapter} 之间`)
      warnings.push(`丢弃越界事件：chapterNo=${String(item.chapterNo)}`)
      continue
    }
    const title = cleanText(item.title, 80)
    if (!title) {
      if (strict) throw new StoryAtlasValidationError('时间线事件标题不能为空')
      warnings.push(`丢弃缺少标题的事件：chapterNo=${chapterNo}`)
      continue
    }
    const timeLabel = cleanText(item.timeLabel, 60)
    const summary = cleanText(item.summary, 500)
    const importance = clampImportance(item.importance)
    const source = TIMELINE_SOURCES.includes(item.source as TimelineSource)
      ? item.source as TimelineSource
      : defaultSource
    const status = TIMELINE_STATUSES.includes(item.status as TimelineStatus)
      ? item.status as TimelineStatus
      : defaultStatus

    let placeId: string | undefined
    const rawPlaceId = typeof item.placeId === 'string' ? item.placeId.trim() : ''
    if (rawPlaceId && TIMELINE_ID.test(rawPlaceId) && (!placeIds || placeIds.has(rawPlaceId))) {
      placeId = rawPlaceId
    } else if (rawPlaceId && placeIds) {
      warnings.push(`地点 ${rawPlaceId} 不存在，已清空 placeId（保留 placeName）`)
    }
    const placeName = cleanText(item.placeName, 80) || undefined
    const characters = Array.isArray(item.characters)
      ? item.characters
          .filter((name): name is string => typeof name === 'string')
          .map((name) => cleanText(name, 40))
          .filter(Boolean)
          .slice(0, 20)
      : undefined

    let id = typeof item.id === 'string' && TIMELINE_ID.test(item.id) ? item.id : ''
    if (!id || usedIds.has(id)) {
      const base = `${idPrefix}-${chapterNo}`
      let suffix = 1
      while (usedIds.has(`${base}-${suffix}`)) suffix += 1
      id = `${base}-${suffix}`
    }
    usedIds.add(id)

    const event: TimelineEvent = {
      id,
      chapterNo,
      timeLabel,
      title,
      summary,
      importance,
      source,
      status,
    }
    if (placeId) event.placeId = placeId
    if (placeName) event.placeName = placeName
    if (characters?.length) event.characters = characters
    events.push(event)
  }

  events.sort((a, b) => a.chapterNo - b.chapterNo)
  return { events, warnings }
}

export function parseTimelineEvents(raw: unknown): TimelineEvent[] {
  if (raw === null || raw === undefined || raw === '') return []
  try {
    return normalizeTimelineEvents(raw).events
  } catch {
    console.warn('[story-atlas] 时间线 JSON 损坏，已降级为空')
    return []
  }
}

export function serializeTimelineEvents(events: TimelineEvent[]): string {
  return JSON.stringify(events)
}

export function mergeChapterTimeline(
  existing: TimelineEvent[],
  chapterNo: number,
  incoming: unknown,
): TimelineEvent[] {
  const next = normalizeTimelineEvents(incoming, {
    fixedChapterNo: chapterNo,
    defaultSource: 'chapter',
    defaultStatus: 'planned',
    idPrefix: `ch${String(chapterNo).padStart(3, '0')}`,
  }).events
  const kept = existing.filter((event) => event.chapterNo !== chapterNo)
  return normalizeTimelineEvents([...kept, ...next], { maxChapter: Number.MAX_SAFE_INTEGER }).events
}

export function buildTimelineContext(
  events: TimelineEvent[],
  chapterNo: number,
  options: { radius?: number; limit?: number } = {},
): string {
  const radius = Math.max(0, options.radius ?? 1)
  const limit = Math.max(1, options.limit ?? 12)
  return events
    .filter((event) => Math.abs(event.chapterNo - chapterNo) <= radius)
    .sort((a, b) => a.chapterNo - b.chapterNo)
    .slice(0, limit)
    .map((event) => {
      const time = event.timeLabel ? `${event.timeLabel}：` : ''
      const summary = event.summary ? `：${event.summary}` : ''
      return `- 第${event.chapterNo}章 ${time}${event.title}${summary}`
    })
    .join('\n')
}
```

```ts
export interface WorldMapRegion {
  id: string
  name: string
  description?: string
  color?: string
}

export interface WorldMapFaction {
  id: string
  name: string
  color?: string
}

export interface WorldMapPlace {
  id: string
  name: string
  regionId?: string
  factionId?: string
  x: number
  y: number
  description?: string
  importance: 1 | 2 | 3 | 4 | 5
}

export type WorldMapRouteType = 'road' | 'sea' | 'air' | 'portal' | 'other'

export interface WorldMapRoute {
  id: string
  from: string
  to: string
  type: WorldMapRouteType
  label?: string
  description?: string
}

export interface WorldMapData {
  version: 1
  generatedAt: string | null
  regions: WorldMapRegion[]
  factions: WorldMapFaction[]
  places: WorldMapPlace[]
  routes: WorldMapRoute[]
}

const ROUTE_TYPES: WorldMapRouteType[] = ['road', 'sea', 'air', 'portal', 'other']

function uniqueId(raw: unknown, prefix: string, index: number, used: Set<string>): string {
  const candidate = typeof raw === 'string' && TIMELINE_ID.test(raw) ? raw : `${prefix}-${index + 1}`
  if (!used.has(candidate)) {
    used.add(candidate)
    return candidate
  }
  let suffix = 2
  while (used.has(`${candidate}-${suffix}`)) suffix += 1
  const next = `${candidate}-${suffix}`
  used.add(next)
  return next
}

function arrayOf(value: unknown, strict: boolean, field: string): unknown[] {
  if (value === null || value === undefined) return []
  if (!Array.isArray(value)) {
    if (strict) throw new StoryAtlasValidationError(`${field} 必须是数组`)
    return []
  }
  return value
}

export function normalizeWorldMap(
  input: unknown,
  options: { strict?: boolean } = {},
): { data: WorldMapData | null; warnings: string[] } {
  const strict = Boolean(options.strict)
  const warnings: string[] = []
  if (input === null || input === undefined) {
    if (strict) throw new StoryAtlasValidationError('world_map 不能为空')
    return { data: null, warnings }
  }
  if (typeof input === 'string') {
    try {
      input = JSON.parse(input)
    } catch {
      if (strict) throw new StoryAtlasValidationError('世界地图 JSON 格式不正确')
      return { data: null, warnings: ['世界地图 JSON 损坏，已降级为空'] }
    }
  }
  if (!input || typeof input !== 'object' || Array.isArray(input)) {
    if (strict) throw new StoryAtlasValidationError('world_map 必须是对象')
    return { data: null, warnings: ['世界地图不是对象，已降级为空'] }
  }
  const root = input as Record<string, unknown>

  const regionIds = new Set<string>()
  const regions = arrayOf(root.regions, strict, 'regions')
    .slice(0, 30)
    .map((raw, index) => {
      const item = (raw && typeof raw === 'object' ? raw : {}) as Record<string, unknown>
      const name = cleanText(item.name, 80)
      if (!name) return null
      const region: WorldMapRegion = { id: uniqueId(item.id, 'region', index, regionIds), name }
      const description = cleanText(item.description, 300)
      const color = cleanText(item.color, 32)
      if (description) region.description = description
      if (color) region.color = color
      return region
    })
    .filter((region): region is WorldMapRegion => Boolean(region))
  if (arrayOf(root.regions, strict, 'regions').length > 30) warnings.push('区域超过 30 个，已截断')

  const factionIds = new Set<string>()
  const factions = arrayOf(root.factions, strict, 'factions')
    .slice(0, 30)
    .map((raw, index) => {
      const item = (raw && typeof raw === 'object' ? raw : {}) as Record<string, unknown>
      const name = cleanText(item.name, 80)
      if (!name) return null
      const faction: WorldMapFaction = { id: uniqueId(item.id, 'faction', index, factionIds), name }
      const color = cleanText(item.color, 32)
      if (color) faction.color = color
      return faction
    })
    .filter((faction): faction is WorldMapFaction => Boolean(faction))
  if (arrayOf(root.factions, strict, 'factions').length > 30) warnings.push('势力超过 30 个，已截断')

  const rawPlaces = arrayOf(root.places, strict, 'places').slice(0, 200)
  if (arrayOf(root.places, strict, 'places').length > 200) warnings.push('地点超过 200 个，已截断')
  const placeIds = new Set<string>()
  const validPlaces = rawPlaces.map((raw, index) => {
    const item = (raw && typeof raw === 'object' ? raw : {}) as Record<string, unknown>
    const name = cleanText(item.name, 80)
    if (!name) return null
    return { item, name, id: uniqueId(item.id, 'place', index, placeIds) }
  }).filter((value): value is { item: Record<string, unknown>; name: string; id: string } => Boolean(value))
  const columns = Math.max(1, Math.ceil(Math.sqrt(validPlaces.length || 1)))
  const rows = Math.max(1, Math.ceil(validPlaces.length / columns))
  const places = validPlaces.map(({ item, name, id }, index) => {
    const rawX = Number(item.x)
    const rawY = Number(item.y)
    const fallbackX = (index % columns + 0.5) / columns
    const fallbackY = (Math.floor(index / columns) + 0.5) / rows
    const x = Number.isFinite(rawX) ? Math.max(0, Math.min(1, rawX)) : fallbackX
    const y = Number.isFinite(rawY) ? Math.max(0, Math.min(1, rawY)) : fallbackY
    const place: WorldMapPlace = {
      id,
      name,
      x: Math.round(x * 1000) / 1000,
      y: Math.round(y * 1000) / 1000,
      importance: clampImportance(item.importance),
    }
    const regionId = typeof item.regionId === 'string' ? item.regionId : ''
    if (regionId && regionIds.has(regionId)) place.regionId = regionId
    else if (regionId) warnings.push(`地点 ${id} 的区域 ${regionId} 不存在，已清空`)
    const factionId = typeof item.factionId === 'string' ? item.factionId : ''
    if (factionId && factionIds.has(factionId)) place.factionId = factionId
    else if (factionId) warnings.push(`地点 ${id} 的势力 ${factionId} 不存在，已清空`)
    const description = cleanText(item.description, 300)
    if (description) place.description = description
    return place
  })

  const routeIds = new Set<string>()
  const routeKeys = new Set<string>()
  const rawRoutes = arrayOf(root.routes, strict, 'routes').slice(0, 100)
  if (arrayOf(root.routes, strict, 'routes').length > 100) warnings.push('路线超过 100 条，已截断')
  const routes: WorldMapRoute[] = []
  rawRoutes.forEach((raw, index) => {
    const item = (raw && typeof raw === 'object' ? raw : {}) as Record<string, unknown>
    const from = typeof item.from === 'string' ? item.from : ''
    const to = typeof item.to === 'string' ? item.to : ''
    if (!placeIds.has(from) || !placeIds.has(to) || from === to) {
      warnings.push(`丢弃无效路线：${from || '?'} → ${to || '?'}`)
      return
    }
    const key = `${from}|${to}`
    if (routeKeys.has(key)) {
      warnings.push(`丢弃重复路线：${from} → ${to}`)
      return
    }
    routeKeys.add(key)
    const route: WorldMapRoute = {
      id: uniqueId(item.id, 'route', index, routeIds),
      from,
      to,
      type: ROUTE_TYPES.includes(item.type as WorldMapRouteType) ? item.type as WorldMapRouteType : 'other',
    }
    const label = cleanText(item.label, 80)
    const description = cleanText(item.description, 300)
    if (label) route.label = label
    if (description) route.description = description
    routes.push(route)
  })

  const generatedAt = typeof root.generatedAt === 'string' && !Number.isNaN(Date.parse(root.generatedAt))
    ? root.generatedAt
    : null
  return {
    data: { version: 1, generatedAt, regions, factions, places, routes },
    warnings,
  }
}

export function parseWorldMap(raw: unknown): WorldMapData | null {
  if (raw === null || raw === undefined || raw === '') return null
  try {
    return normalizeWorldMap(raw).data
  } catch {
    console.warn('[story-atlas] 世界地图 JSON 损坏，已降级为空')
    return null
  }
}

export function serializeWorldMap(map: WorldMapData): string {
  return JSON.stringify(map)
}

export function buildMapContext(
  map: WorldMapData | null,
  events: TimelineEvent[],
  chapterNo: number,
  options: { limit?: number } = {},
): string {
  if (!map || !map.places.length) return ''
  const limit = Math.max(1, options.limit ?? 8)
  const regionNames = new Map(map.regions.map((region) => [region.id, region.name]))
  const eventPlaceIds = new Set(
    events.filter((event) => event.chapterNo === chapterNo && event.placeId).map((event) => event.placeId as string),
  )
  let selected = map.places.filter((place) => eventPlaceIds.has(place.id))
  if (!selected.length) {
    selected = [...map.places].sort((a, b) => b.importance - a.importance)
  }
  return selected.slice(0, limit)
    .map((place) => {
      const region = place.regionId ? regionNames.get(place.regionId) : ''
      const prefix = region ? `${place.name}（${region}）` : place.name
      return `- ${prefix}${place.description ? `：${place.description}` : ''}`
    })
    .join('\n')
}

export function parseModelJson(text: string): unknown {
  const stripped = text.replace(/```(?:json)?/gi, '').replace(/```/g, '').trim()
  const start = stripped.indexOf('{')
  const end = stripped.lastIndexOf('}')
  if (start < 0 || end <= start) throw new StoryAtlasValidationError('模型返回的 JSON 格式不正确')
  try {
    return JSON.parse(stripped.slice(start, end + 1))
  } catch {
    throw new StoryAtlasValidationError('模型返回的 JSON 格式不正确')
  }
}

export function readStoryAtlas(novel: {
  timelineEvents?: unknown
  worldMap?: unknown
}): { timelineEvents: TimelineEvent[]; worldMap: WorldMapData | null } {
  return {
    timelineEvents: parseTimelineEvents(novel.timelineEvents),
    worldMap: parseWorldMap(novel.worldMap),
  }
}
```

```ts
import { now } from '../../utils/response.js'
import {
  StoryAtlasValidationError,
  normalizeTimelineEvents,
  normalizeWorldMap,
  parseTimelineEvents,
  parseWorldMap,
  serializeTimelineEvents,
  serializeWorldMap,
  type TimelineEvent,
  type WorldMapData,
} from './story-atlas.js'

export class StoryAtlasConflictError extends Error {
  readonly currentUpdatedAt: string | null
  constructor(currentUpdatedAt: string | null) {
    super('小说已被其他任务更新，请刷新后重试')
    this.name = 'StoryAtlasConflictError'
    this.currentUpdatedAt = currentUpdatedAt
  }
}

export class StoryAtlasNotFoundError extends Error {
  constructor() {
    super('小说不存在')
    this.name = 'StoryAtlasNotFoundError'
  }
}

function activeRow(novelId: number) {
  return db.select().from(schema.novels)
    .where(and(eq(schema.novels.id, novelId), isNull(schema.novels.deletedAt)))
    .get()
}

function requireRow(novelId: number) {
  const row = activeRow(novelId)
  if (!row) throw new StoryAtlasNotFoundError()
  return row
}

export function writeTimelineEvents(novelId: number, events: TimelineEvent[]): void {
  requireRow(novelId)
  db.update(schema.novels)
    .set({ timelineEvents: serializeTimelineEvents(events), updatedAt: now() })
    .where(eq(schema.novels.id, novelId))
    .run()
}

export function writeWorldMap(novelId: number, map: WorldMapData): void {
  requireRow(novelId)
  db.update(schema.novels)
    .set({ worldMap: serializeWorldMap(map), updatedAt: now() })
    .where(eq(schema.novels.id, novelId))
    .run()
}

export function saveStoryAtlas(
  novelId: number,
  patch: { timelineEvents?: unknown; worldMap?: unknown; baseUpdatedAt?: string | null },
) {
  if (patch.timelineEvents === undefined && patch.worldMap === undefined) {
    throw new StoryAtlasValidationError('timeline_events 或 world_map 至少提供一个')
  }
  return db.transaction((tx) => {
    const row = tx.select().from(schema.novels)
      .where(and(eq(schema.novels.id, novelId), isNull(schema.novels.deletedAt)))
      .get()
    if (!row) throw new StoryAtlasNotFoundError()
    if (patch.baseUpdatedAt !== undefined && patch.baseUpdatedAt !== null && patch.baseUpdatedAt !== row.updatedAt) {
      throw new StoryAtlasConflictError(row.updatedAt)
    }

    const warnings: string[] = []
    const currentMap = parseWorldMap(row.worldMap)
    let nextMap = currentMap
    if (patch.worldMap !== undefined) {
      const normalized = normalizeWorldMap(patch.worldMap, { strict: true })
      nextMap = normalized.data
      warnings.push(...normalized.warnings)
    }
    const placeIds = nextMap?.places.map((place) => place.id) ?? []

    let nextTimeline = parseTimelineEvents(row.timelineEvents)
    if (patch.timelineEvents !== undefined) {
      const normalized = normalizeTimelineEvents(patch.timelineEvents, {
        strict: true,
        maxChapter: row.numberOfChapters ?? 100,
        placeIds,
      })
      nextTimeline = normalized.events
      warnings.push(...normalized.warnings)
    } else if (patch.worldMap !== undefined) {
      const normalized = normalizeTimelineEvents(nextTimeline, {
        maxChapter: row.numberOfChapters ?? 100,
        placeIds,
      })
      nextTimeline = normalized.events
      warnings.push(...normalized.warnings)
    }

    const updatedAt = now()
    tx.update(schema.novels).set({
      timelineEvents: serializeTimelineEvents(nextTimeline),
      worldMap: nextMap ? serializeWorldMap(nextMap) : null,
      updatedAt,
    }).where(eq(schema.novels.id, novelId)).run()

    return { timelineEvents: nextTimeline, worldMap: nextMap, updatedAt, warnings }
  })
}
```

```ts
export function normalizeNarration(value: unknown): string {
  return typeof value === 'string' ? value.replace(/\s+/g, ' ').trim().slice(0, 500) : ''
}

export function updateChapterNarration(
  blueprint: string | null | undefined,
  chapterNo: number,
  narration: string,
): string | null {
  const text = blueprint || ''
  const block = splitChapterBlueprintBlocks(text).find((item) => item.number === chapterNo)
  if (!block) return null
  const clean = normalizeNarration(narration)
  const lines = block.text
    .split('\n')
    .filter((line) => !/^\s*旁白(?:描述|说明)?\s*[：:]/.test(line))
  if (clean) {
    const anchor = lines.findIndex((line) => /^\s*本章简述\s*[：:]/.test(line))
    const narrationLine = `旁白描述：${clean}`
    if (anchor >= 0) lines.splice(anchor + 1, 0, narrationLine)
    else if (lines.length && !lines[lines.length - 1].trim()) lines.splice(lines.length - 1, 0, narrationLine)
    else lines.push(narrationLine)
  }
  const start = text.indexOf(block.text)
  return `${text.slice(0, start)}${lines.join('\n')}${text.slice(start + block.text.length)}`
}
```

```ts
app.put('/:id/story-atlas', async (c) => {
  const id = parseId(c.req.param('id'))
  if (!id) return badRequest(c, '小说编号无效')
  const body = await readBody(c)
  if (!body) return badRequest(c, '请求体必须是对象')
  const timelineEvents = body.timeline_events ?? body.timelineEvents
  const worldMap = body.world_map ?? body.worldMap
  const baseUpdatedAt = body.base_updated_at ?? body.baseUpdatedAt
  if (timelineEvents === undefined && worldMap === undefined) {
    return badRequest(c, 'timeline_events 或 world_map 至少提供一个')
  }
  try {
    const saved = saveStoryAtlas(id, { timelineEvents, worldMap, baseUpdatedAt })
    return success(c, {
      timeline_events: saved.timelineEvents,
      world_map: saved.worldMap,
      updated_at: saved.updatedAt,
      warnings: saved.warnings,
    })
  } catch (error) {
    if (error instanceof StoryAtlasNotFoundError) return notFound(c, error.message)
    if (error instanceof StoryAtlasConflictError) {
      return c.json({
        code: 409,
        message: error.message,
        data: { current_updated_at: error.currentUpdatedAt },
      }, 409)
    }
    if (error instanceof StoryAtlasValidationError) return badRequest(c, error.message)
    throw error
  }
})

app.patch('/:id/blueprint/chapters/:chapterNo', async (c) => {
  const id = parseId(c.req.param('id'))
  if (!id) return badRequest(c, '小说编号无效')
  const chapterNo = parsePositiveInt(c.req.param('chapterNo'))
  if (!chapterNo) return badRequest(c, '章节号必须是正整数')
  const body = await readBody(c)
  if (!body) return badRequest(c, '请求体必须是对象')
  if (typeof body.narration !== 'string') return badRequest(c, 'narration 必须是字符串')
  const narration = body.narration.replace(/\s+/g, ' ').trim()
  if (narration.length > 500) return badRequest(c, 'narration 不能超过 500 字')
  const detail = getNovel(id)
  if (!detail) return notFound(c, '小说不存在')
  const next = updateChapterNarration(detail.chapterBlueprint, chapterNo, narration)
  if (next === null) return notFound(c, '章节不存在')
  const updated = updateNovel(id, { chapter_blueprint: next })
  const chapter = parseChapterBlueprint(next).find((item) => item.number === chapterNo)
  return success(c, { chapter, chapter_blueprint: next, updated_at: updated?.updatedAt })
```

```ts
  prompt: string,
  kind: 'worldMap' | 'timeline',

    data.generatedAt = now()
```

```ts
async function runAtlas(ctx: JobRunContext): Promise<Record<string, unknown>> {
  let novel = getNovel(ctx.novelId)
  if (!novel) throw new NovelJobError('not_found', '小说不存在', 404)
  const steps: { key: string; skipped: boolean; characters: number }[] = []
  let mapExecuted = false

  for (let index = 0; index < 2; index += 1) {
    const key = index === 0 ? 'worldMap' : 'timeline'
    const step = ARCHITECTURE_STEPS.find((item) => item.key === key)
    if (!step || step.kind !== 'atlas') throw new NovelJobError('protocol', '故事坐标步骤缺失')
    const hasMap = Boolean(novel.worldMap?.places.length)
    const hasTimeline = Boolean(novel.timelineEvents.length)

    const skip = !ctx.force && existing && !(key === 'timeline' && mapExecuted)
    if (skip) {
```

```ts
async function updateChapterTimeline(
  ctx: JobRunContext,
  novel: NovelDetail,
  chapterNo: number,
  content: string,
  status: 'planned' | 'confirmed',
): Promise<{ timelineUpdated: boolean; timelineError?: string }> {
  try {
    beginStage(ctx, 'timeline', '正在更新时间事件线...', status === 'confirmed' ? 3 : 2, status === 'confirmed' ? 3 : 2)
    const existing = novel.timelineEvents.filter((event) => event.chapterNo === chapterNo)
    const neighbors = buildTimelineContext(novel.timelineEvents, chapterNo, { radius: 1, limit: 10 })
    const prompt = chapterPrompts.chapterTimeline({
      chapterNo,
      chapterText: content,
      existingEvents: JSON.stringify(existing),
      worldMapJson: novel.worldMap ? JSON.stringify(novel.worldMap) : '{}',
      neighborEvents: neighbors,
    })
    const output = await ctx.llm.complete(prompt, { signal: ctx.controller.signal })
    assertActive(ctx)
    const parsed = parseModelJson(output) as Record<string, unknown>
    const { events } = normalizeTimelineEvents(parsed.events, {
      strict: true,
      fixedChapterNo: chapterNo,
      maxChapter: novel.numberOfChapters ?? 100,
      defaultSource: 'chapter',
      defaultStatus: status,
      idPrefix: `ch${String(chapterNo).padStart(3, '0')}`,
      placeIds: novel.worldMap?.places.map((place) => place.id) ?? [],
    })
    writeTimelineEvents(ctx.novelId, mergeChapterTimeline(novel.timelineEvents, chapterNo, events))
    return { timelineUpdated: true }
  } catch (error) {
    if (error instanceof NovelJobCancelled) throw error
    // 超时保留原始错误，交给 handleFailure 映射为 timeout，而不是误记成 cancelled
    if (ctx.timedOut) throw error
    if (ctx.controller.signal.aborted || isCancelRequested(ctx.jobId)) throw new NovelJobCancelled()
    const message = error instanceof StoryAtlasValidationError || error instanceof NovelJobError
      ? error.message
      : '时间线更新失败'
    return { timelineUpdated: false, timelineError: message }
  }
}
```

```ts
export interface TimelineEventLike {
  id: string
  chapterNo: number
  title: string
  placeId?: string
  status?: string
  [key: string]: any
}

export interface WorldMapLike {
  version: number
  generatedAt: string | null
  regions: Array<{ id: string; name: string; color?: string }>
  factions: Array<{ id: string; name: string; color?: string }>
  places: Array<{ id: string; name: string; regionId?: string; factionId?: string; x: number; y: number; importance: number; description?: string }>
  routes: Array<{ id: string; from: string; to: string; type: string; label?: string }>
}

export function filterTimelineEvents<T extends TimelineEventLike>(
  events: T[],
  options: { chapterNo?: number; scope?: 'all' | 'chapter' | 'confirmed' } = {},
): T[] {
  const { chapterNo = 0, scope = 'all' } = options
  return events.filter((event) => {
    if (scope === 'chapter') return event.chapterNo === chapterNo
    if (scope === 'confirmed') return event.status === 'confirmed'
    return true
  })
}

export function eventPlaceIds(events: TimelineEventLike[], chapterNo: number): Set<string> {
  return new Set(events
    .filter((event) => event.chapterNo === chapterNo && event.placeId)
    .map((event) => event.placeId as string))
}

export function buildMapGraphOption(
  mapData: WorldMapLike | null | undefined,
  options: { highlightIds?: Set<string>; dark?: boolean } = {},
) {
  const highlight = options.highlightIds || new Set<string>()
  const dark = Boolean(options.dark)
  const factionColors = new Map((mapData?.factions || []).map((faction) => [faction.id, faction.color]))
  const regionColors = new Map((mapData?.regions || []).map((region) => [region.id, region.color]))
  const data = (mapData?.places || []).map((place) => ({
    id: place.id,
    name: place.name,
    x: place.x * 1000,
    y: place.y * 600,
    symbolSize: 10 + place.importance * 4,
    itemStyle: {
      color: (place.factionId && factionColors.get(place.factionId))
        || (place.regionId && regionColors.get(place.regionId))
        || (dark ? '#8a94a6' : '#6b7280'),
      borderColor: highlight.has(place.id) ? '#f5a524' : 'transparent',
      borderWidth: highlight.has(place.id) ? 3 : 0,
    },
    label: { show: true, color: dark ? '#e5e7eb' : '#1f2937', fontSize: 11 },
    _raw: place,
  }))
  const links = (mapData?.routes || []).map((route) => ({
    source: route.from,
    target: route.to,
    label: { show: Boolean(route.label), formatter: route.label || '' },
    lineStyle: {
      type: route.type === 'sea' || route.type === 'air' ? 'dashed' : 'solid',
      color: dark ? '#6b7280' : '#9ca3af',
      curveness: 0.12,
    },
    _raw: route,
  }))
  return {
    backgroundColor: 'transparent',
    tooltip: { trigger: 'item', confine: true },
    series: [{
      type: 'graph',
      layout: 'none',
      roam: true,
      draggable: true,
      data,
      links,
      edgeSymbol: ['none', 'arrow'],
      edgeSymbolSize: 6,
      lineStyle: { color: dark ? '#6b7280' : '#9ca3af' },
    }],
  }
}
```
